```python
# tests/legal_search/regression/test_citation_integrity.py

class TestCitationIntegrity:
    """
    Regression suite untuk memastikan pencarian hukum tidak kehilangan:
    - Kutipan asli (verbatim)
    - Identitas sumber (UU, PP, Permen, Yurisprudensi)
    - Checksum dokumen
    - Peringatan keberlakuan (masih berlaku / dicabut / diamendemen)
    """
    
    def test_citation_preserved_after_reindex(self):
        """Kutipan harus identik sebelum & sesudah re-indexing"""
        before = search("wanprestasi kontraktor", top_k=10)
        reindex_collection()
        after = search("wanprestasi kontraktor", top_k=10)
        
        for b, a in zip(before, after):
            assert b.quote == a.quote, "Kutipan berubah setelah re-index"
            assert b.source_id == a.source_id
            assert b.checksum == a.checksum
            assert b.validity_status == a.validity_status
    
    def test_amended_regulation_flagged(self):
        """Regulasi yang diamendemen harus menampilkan peringatan"""
        results = search("denda keterlambatan 1‰")
        # Permen PUPR 14/2020 sudah diamendemen sebagian
        amended_hit = [r for r in results if r.source_id == "PERMEN_PUPR_14_2020"]
        assert all(r.validity_status == "AMENDED" for r in amended_hit)
        assert all(r.amendment_warning is not None for r in amended_hit)
    
    def test_revoked_regulation_excluded_or_flagged(self):
        """Regulasi yang dicabut tidak muncul di hasil utama"""
        results = search("standar lama konstruksi 2010")
        revoked = [r for r in results if r.validity_status == "REVOKED"]
        assert len(revoked) == 0 or all(r.is_shadow_result for r in revoked)
    
    def test_chunk_boundary_preserves_context(self):
        """Pemotongan konteks tidak boleh memutus kalimat atau pasal"""
        results = search("tanggung jawab kegagalan bangunan")
        for r in results:
            # Chunk tidak boleh dimulai/tengah kalimat terpotong
            assert not r.quote.startswith("dan ")
            assert not r.quote.startswith("yang ")
            assert r.quote.strip().endswith((".", ")", "\""))
    
    def test_collection_growth_no_degradation(self, benchmark):
        """Performa pencarian tidak turun saat koleksi tumbuh 10x"""
        small_collection = load_collection(size=1_000)
        large_collection = load_collection(size=10_000)
        
        queries = load_golden_queries()  # 100 query + ground truth
        
        small_latency = benchmark(small_collection, queries)
        large_latency = benchmark(large_collection, queries)
        
        # Latency naik maksimal 30% saat koleksi 10x lipat
        assert large_latency.p95 <= small_latency.p95 * 1.3
        # Recall tidak turun lebih dari 2%
        assert large_latency.recall >= small_latency.recall - 0.02
```

```python
# services/notifications/time_bar_reminder.py

class TimeBarReminder:
    """
    Strategi eskalasi bertahap untuk notice deadlines.
    Berdasarkan psikologi: semakin dekat deadline, semakin urgent & frequent.
    """
    
    ESCALATION_SCHEDULE = {
        # (days_before, channel, recipients, tone)
        30: {
            "channels": ["email"],
            "recipients": ["contract_admin"],
            "tone": "INFO",
            "template": "deadline_30d_info"
        },
        14: {
            "channels": ["email", "in_app"],
            "recipients": ["contract_admin", "project_manager"],
            "tone": "WARNING",
            "template": "deadline_14d_warning"
        },
        7: {
            "channels": ["email", "in_app", "whatsapp"],
            "recipients": ["contract_admin", "project_manager", "legal"],
            "tone": "URGENT",
            "template": "deadline_7d_urgent"
        },
        3: {
            "channels": ["email", "in_app", "whatsapp", "sms"],
            "recipients": ["contract_admin", "project_manager", "legal", "director"],
            "tone": "CRITICAL",
            "template": "deadline_3d_critical"
        },
        1: {
            "channels": ["email", "in_app", "whatsapp", "sms", "phone_call"],
            "recipients": ["contract_admin", "project_manager", "legal", "director", "ceo"],
            "tone": "EMERGENCY",
            "template": "deadline_1d_emergency"
        },
        0: {  # Hari H
            "channels": ["email", "in_app", "whatsapp", "sms", "phone_call"],
            "recipients": ["ALL_STAKEHOLDERS"],
            "tone": "FINAL",
            "template": "deadline_today_final"
        }
    }
    
    async def process_deadline(self, notice_id: str):
        notice = await get_notice(notice_id)
        days_left = (notice.deadline - datetime.now()).days
        
        if days_left < 0:
            await self._handle_expired(notice)
            return
        
        schedule = self._get_schedule_for(days_left)
        if not schedule:
            return
        
        message = await self._render_message(notice, schedule)
        
        for channel in schedule["channels"]:
            for recipient in schedule["recipients"]:
                await self._send_via_channel(channel, recipient, message)
        
        await self._log_reminder(notice_id, days_left, schedule)
    
    async def _handle_expired(self, notice):
        """Time bar sudah berlaku — tandai hak klaim hangus"""
        await mark_claim_barred(notice.id)
        await notify_all_parties(
            notice, 
            tone="BARRED",
            message=f"Hak klaim atas notice {notice.id} telah hangus "
                    f"karena tidak diajukan dalam tenggat waktu."
        )
        await create_audit_event("CLAIM_BARRED", notice.id)
```

### 🎯 Task #15: AI Root Cause Analysis untuk Kegagalan Bangunan (PRIORITAS TINGGI)

**Mengapa penting:** Modul Building Failure Forensics butuh AI untuk menganalisis penyebab kegagalan secara otomatis berdasarkan data lapangan.

**Model pipeline:**

```python
# services/forensics/root_cause_analyzer.py

class RootCauseAnalyzer:
    """
    Multi-modal AI untuk analisis penyebab kegagalan bangunan.
    Input: Foto retak, data sensor IoT, hasil uji lab, daily reports
    Output: Probabilitas penyebab + confidence score
    """
    
    CAUSE_CATEGORIES = [
        "DESIGN_ERROR",           # Kesalahan perencana
        "CONSTRUCTION_DEFECT",    # Cacat pelaksanaan
        "SUPERVISION_FAILURE",    # Kegagalan pengawasan
        "MATERIAL_FAILURE",       # Material di bawah standar
        "FORCE_MAJEURE",          # Bencana alam
        "USER_MISUSE",            # Penggunaan tidak sesuai
        "NORMAL_WEAR",            #ausa normal
        "UNKNOWN"
    ]
    
    async def analyze(self, failure_case_id: str) -> RootCauseReport:
        case = await get_failure_case(failure_case_id)
        
        # 1. Visual analysis dari foto retak
        visual_features = await self._analyze_photos(case.photos)
        # Pattern recognition: retak diagonal → shear failure, 
        # retak horizontal → settlement, dll
        
        # 2. Sensor data analysis (jika ada IoT)
        sensor_features = await self._analyze_sensor_data(case.sensor_readings)
        
        # 3. Lab results analysis
        lab_features = await self._analyze_lab_results(case.lab_tests)
        
        # 4. Context analysis (cuaca, timeline, daily reports)
        context_features = await self._analyze_context(case)
        
        # 5. Multi-modal fusion
        combined_features = self._fuse_features(
            visual_features, sensor_features, 
            lab_features, context_features
        )
        
        # 6. Classification
        cause_probabilities = self.classifier.predict_proba(combined_features)
        
        # 7. Liability apportionment (berdasarkan kontrak)
        liability = await self._calculate_liability(
            case.contract_id, cause_probabilities
        )
        
        return RootCauseReport(
            primary_cause=max(cause_probabilities, key=cause_probabilities.get),
            probabilities=cause_probabilities,
            confidence=max(cause_probabilities.values()),
            liability_apportionment=liability,
            evidence_chain=self._build_evidence_chain(case),
            recommendations=self._generate_recommendations(cause_probabilities)
        )
```

### 🎯 Task #16: Settlement Recommender Engine (PRIORITAS SEDANG)

**Mengapa penting:** Memberikan rekomendasi penyelesaian konstruktif berbasis data, bukan emosi.

**Algorithm:**

```python
# services/dispute/settlement_recommender.py

class SettlementRecommender:
    """
    Rekomendasi penyelesaian berdasarkan:
    - BATNA (Best Alternative to Negotiated Agreement) masing-masing pihak
    - ZOPA (Zone of Possible Agreement)
    - Historical settlement data
    - Relationship preservation score
    """
    
    async def recommend(self, matter_id: str) -> SettlementRecommendation:
        matter = await get_matter(matter_id)
        
        # 1. Hitung BATNA masing-masing pihak
        claimant_batna = await self._calculate_batna(
            matter, party="claimant", 
            alternative="arbitration"  # atau "litigation"
        )
        respondent_batna = await self._calculate_batna(
            matter, party="respondent",
            alternative="arbitration"
        )
        
        # 2. Identifikasi ZOPA
        zopa = self._calculate_zopa(claimant_batna, respondent_batna)
        
        if zopa.is_empty:
            return SettlementRecommendation(
                feasible=False,
                reason="Tidak ada zona kesepakatan. Rekomendasi: adjudikasi.",
                suggested_expert_type="SENIOR_ARBITRATOR"
            )
        
        # 3. Cari settlement serupa dari database
        similar_cases = await self._find_similar_settlements(matter, top_k=10)
        
        # 4. Hitung Nash Bargaining Solution
        nash_point = self._nash_bargaining(
            claimant_batna, respondent_batna, zopa
        )
        
        # 5. Adjust untuk relationship preservation
        if matter.preserve_relationship:
            nash_point = self._adjust_for_relationship(nash_point, similar_cases)
        
        # 6. Generate opsi-opsi kreatif (non-monetary)
        creative_options = await self._generate_creative_options(matter)
        
        return SettlementRecommendation(
            feasible=True,
            recommended_settlement=nash_point,
            zopa=zopa,
            alternative_options=creative_options,
            supporting_cases=similar_cases,
            confidence=self._calculate_confidence(similar_cases),
            suggested_mediator=self._recommend_mediator(matter)
        )
```
