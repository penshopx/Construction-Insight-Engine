### 1. COUNSEL_ONLY — Definisi yang Jelas

**Masalah:** Matriks hanya mengizinkan counsel, tapi nama mengimplikasikan "penasihat saja". Jika klien juga harus akses, matriks salah.

**Perbaikan:** Definisikan dengan eksplisit.

```python
# Klasifikasi akses yang diperbaiki
ACCESS_CLASSES = {
    "MATTER_PUBLIC": "Semua peserta matter",
    "MATTER_INTERNAL": "Peserta dari organisasi yang sama",
    "RESTRICTED": "Peserta yang dieksplisitkan di restricted_to",
    "MEDIATOR_EYES": "Hanya mediator",
    "COUNSEL_ONLY": "Hanya counsel dari pihak yang meng-upload (klien TIDAK termasuk)",
    "COUNSEL_AND_CLIENT": "Counsel + klien dari pihak yang meng-upload"
}

# Matriks izin yang diperbaiki
def evaluate_access(doc, participant):
    if doc.access_class == "COUNSEL_ONLY":
        # Hanya counsel, klien TIDAK bisa akses
        return participant.role in [
            "CLAIMANT_COUNSEL" if doc.counsel_party == "CLAIMANT" else None,
            "RESPONDENT_COUNSEL" if doc.counsel_party == "RESPONDENT" else None
        ] and participant.role is not None
    
    if doc.access_class == "COUNSEL_AND_CLIENT":
        # Counsel + klien
        is_matching_counsel = (
            (doc.counsel_party == "CLAIMANT" and participant.role == "CLAIMANT_COUNSEL") or
            (doc.counsel_party == "RESPONDENT" and participant.role == "RESPONDENT_COUNSEL")
        )
        is_matching_client = (
            (doc.counsel_party == "CLAIMANT" and participant.role == "CLAIMANT") or
            (doc.counsel_party == "RESPONDENT" and participant.role == "RESPONDENT")
        )
        return is_matching_counsel or is_matching_client
```

**Dokumentasi untuk user:**
```
COUNSEL_ONLY: Dokumen khusus penasihat hukum. Klien tidak bisa mengakses.
COUNSEL_AND_CLIENT: Dokumen untuk penasihat dan klien. Pihak lawan tidak bisa mengakses.

### 2. RLS Rekursif — Solusi yang Aman

**Masalah:** Policy membaca view yang membaca table → rekursi.

**Perbaikan:** ACL table terpisah, bukan view.

-- Tabel ACL eksplisit (bukan view)
CREATE TABLE document_acl (
    document_id UUID REFERENCES documents(id),
    participant_id UUID REFERENCES mediation_participants(id),
    can_access BOOLEAN NOT NULL,
    granted_at TIMESTAMPTZ DEFAULT NOW(),
    PRIMARY KEY (document_id, participant_id)
);

-- Fungsi yang menghitung ACL (bukan recursive view)
CREATE OR REPLACE FUNCTION check_document_access(
    p_document_id UUID,
    p_user_id UUID
) RETURNS BOOLEAN AS $$
DECLARE
    v_can_access BOOLEAN;
BEGIN
    -- Langsung query ACL table, tidak lewat view
    SELECT can_access INTO v_can_access
    FROM document_acl
    WHERE document_id = p_document_id
      AND participant_id IN (
          SELECT id FROM mediation_participants 
          WHERE user_id = p_user_id AND status = 'ACTIVE'
      );
    
    RETURN COALESCE(v_can_access, false);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- RLS menggunakan fungsi, bukan view
CREATE POLICY documents_select_policy ON documents
    FOR SELECT
    USING (
        check_document_access(
            id, 
            current_setting('app.current_user_id')::uuid
        )
    );
```

**Keuntungan:**
- Tidak ada rekursi
- ACL bisa di-cache untuk performa
- Audit trail untuk perubahan ACL

### 3. Konteks User Database — Middleware Wajib

**Masalah:** `current_setting('app.current_user_id')` bisa bocor di connection pool.

**Perbaikan:** Middleware yang selalu set dan reset.

```python
# middleware/database_context.py

class DatabaseContextMiddleware:
    """
    Middleware yang memastikan setiap request memiliki konteks user yang benar.
    Mencegah kebocoran identitas di connection pool.
    """
    
    async def __call__(self, request, call_next):
        # 1. Extract user dari token
        user = await extract_user_from_token(request)
        
        # 2. Get connection dari pool
        conn = await self.pool.acquire()
        
        try:
            # 3. Set user context
            await conn.execute(
                "SELECT set_config('app.current_user_id', $1, true)",
                str(user.id)
            )
            
            # 4. Set organization context (untuk filter MATTER_INTERNAL)
            await conn.execute(
                "SELECT set_config('app.current_org_id', $1, true)",
                str(user.organization_id)
            )
            
            # 5. Attach connection ke request
            request.state.db = conn
            
            # 6. Process request
            response = await call_next(request)
            
            return response
        
        finally:
            # 7. Reset context SEBELUM release connection
            await conn.execute("SELECT set_config('app.current_user_id', '', true)")
            await conn.execute("SELECT set_config('app.current_org_id', '', true)")
            
            # 8. Release connection
            await self.pool.release(conn)


# Test untuk memastikan tidak ada kebocoran
async def test_no_context_leakage():
    """Konteks user tidak bocor ke request berikutnya"""
    
    # Request 1: User A
    response1 = await client.get(
        "/documents/doc-1",
        headers=auth_header(user_a)
    )
    assert response1.status_code == 200
    
    # Request 2: User B (harus tidak bisa akses doc-1)
    response2 = await client.get(
        "/documents/doc-1",
        headers=auth_header(user_b)
    )
    assert response2.status_code == 404  # Bukan 200
    
    # Verifikasi: context di-reset
    conn = await pool.acquire()
    try:
        user_id = await conn.fetchval("SELECT current_setting('app.current_user_id')")
        assert user_id == ""  # Harus kosong setelah reset
    finally:
        await pool.release(conn)

### 4. Legal Hold — Versi Koreksi dengan Audit Trail

**Masalah:** Legal hold tidak selalu berarti data tidak boleh diperbaiki. Koreksi metadata mungkin perlu.

**Perbaikan:** Versioning, bukan freeze total.

```python
# services/legal/legal_hold_service.py

class LegalHoldService:
    
    async def update_document(self, document_id: str, updates: dict, user_id: str):
        """
        Update dokumen yang di-hold.
        Tidak boleh delete atau destructive overwrite.
        Boleh koreksi metadata dengan versi baru.
        """
        # 1. Check legal hold
        hold = await self.get_active_hold_for_document(document_id)
        
        if hold:
            # 2. Validasi update bukan destructive
            if self._is_destructive_update(updates):
                raise LegalHoldViolation(
                    "Dokumen sedang di-hold. Destructive update tidak diizinkan."
                )
            
            # 3. Buat versi baru (bukan overwrite)
            new_version = await self.create_version(
                document_id=document_id,
                updates=updates,
                previous_version=document.current_version,
                reason="correction_under_legal_hold",
                updated_by=user_id
            )
            
            # 4. Audit trail ketat
            await self.audit.log(
                action="document_updated_under_legal_hold",
                document_id=document_id,
                version=new_version.version_number,
                metadata={
                    "hold_id": hold.id,
                    "updates": updates,
                    "updated_by": user_id,
                    "reason": "correction_under_legal_hold"
                },
                severity="HIGH"
            )
            
            return new_version
        
        else:
            # Tidak ada hold, update normal
            return await self.normal_update(document_id, updates, user_id)
    
    def _is_destructive_update(self, updates: dict) -> bool:
        """Check apakah update bersifat destructive"""
        destructive_fields = [
            "file_content",  # Tidak boleh ganti file
            "file_checksum",  # Tidak boleh ganti checksum
            "deleted_at"  # Tidak boleh soft-delete
        ]
        
        return any(field in updates for field in destructive_fields)
```

**Prinsip:**
- Legal hold mencegah **penghapusan** dan **destructive overwrite**
- Legal hold mengizinkan **koreksi metadata** dengan versi baru
- Semua update di bawah hold tercatat di audit trail dengan severity HIGH

**Masalah:** Escrow key berarti platform bisa decrypt → bukan zero-knowledge.

```markdown
## Threat Model — Enkripsi Dokumen COUNSEL_AND_CLIENT

### Pihak yang Terlibat
1. **Platform (KontrakGuard)** — Menyimpan dokumen, menyediakan akses
2. **Counsel** — Penasihat hukum, pemegang private key
3. **Client** — Klien counsel, pemegang private key
4. **Court** — Pengadilan, bisa meminta decrypt via court order
5. **Attacker** — Pihak yang ingin mengakses dokumen tanpa izin

**Opsi A: Platform memegang escrow key**

- Keuntungan: Platform tidak bisa decrypt tanpa persetujuan pihak ketiga
- Kerugian: Proses court order lebih lambat
- Threat model: Zero-knowledge dari platform

### 7. Risk Score — Metodologi yang Lebih Baik

**Masalah:** Average bisa menurunkan skor, geografi 500km bisa false positive.

**Perbaikan:** Weighted scoring dengan threshold per sinyal.

```python
# services/rate_limiting/risk_scorer.py

class RiskScorer:
    """
    Risk scoring dengan metodologi yang lebih baik.
    Bukan average, tapi weighted sum dengan threshold per sinyal.
    """
    
    # Sinyal dengan bobot dan threshold
    SIGNALS = {
        "volume_anomaly": {
            "weight": 0.4,
            "threshold": lambda user: get_user_avg_daily(user) * 3,
            "scoring": "binary"  # 0 atau 1
        },
        "new_device": {
            "weight": 0.3,
            "threshold": None,  # Selalu trigger jika perangkat baru
            "scoring": "binary"
        },
        "geographic_anomaly": {
            "weight": 0.3,
            "threshold": None,
            "scoring": "distance",  # Skor berdasarkan jarak
            "params": {
                "base_distance_km": 100,  # Normal
                "max_distance_km": 1000,  # Maksimum skor
                "vpn_penalty": 0.5  # Kurangi skor jika terdeteksi VPN
            }
        },
        "off_hours": {
            "weight": 0.2,
            "threshold": None,
            "scoring": "time_based",  # Skor berdasarkan jam
            "params": {
                "work_hours": (8, 18),  # 08:00 - 18:00
                "weekend_penalty": 0.3
            }
        },
        "rapid_sequential": {
            "weight": 0.3,
            "threshold": 50,  # 50 dokumen dalam 5 menit
            "scoring": "binary"
        }
    }
    
    async def calculate_risk_score(self, user_id: str, action: str) -> float:
        """Hitung risk score dengan weighted sum"""
        
        total_score = 0.0
        triggered_signals = []
        
        for signal_name, signal_config in self.SIGNALS.items():
            signal_value = await self._evaluate_signal(user_id, signal_name, signal_config)
            
            if signal_value > 0:
                weighted_score = signal_value * signal_config["weight"]
                total_score += weighted_score
                triggered_signals.append({
                    "signal": signal_name,
                    "value": signal_value,
                    "weighted_score": weighted_score
                })
        
        # Cap di 1.0
        final_score = min(total_score, 1.0)
        
        # Log sinyal yang trigger
        await self.audit.log(
            user_id=user_id,
            action="risk_score_calculated",
            metadata={
                "final_score": final_score,
                "triggered_signals": triggered_signals
            }
        )
        
        return final_score
    
    async def _evaluate_signal(self, user_id: str, signal_name: str, config: dict) -> float:
        """Evaluasi satu sinyal"""
        
        if signal_name == "volume_anomaly":
            threshold = config["threshold"](user_id)
            current = await self.get_current_count(user_id, "day")
            return 1.0 if current > threshold else 0.0
        
        elif signal_name == "new_device":
            is_new = await self.is_new_device(user_id)
            return 1.0 if is_new else 0.0
        
        elif signal_name == "geographic_anomaly":
            distance_km = await self.get_distance_from_baseline(user_id)
            
            # Skor berdasarkan jarak
            base = config["params"]["base_distance_km"]
            max_dist = config["params"]["max_distance_km"]
            
            if distance_km <= base:
                score = 0.0
            elif distance_km >= max_dist:
                score = 1.0
            else:
                score = (distance_km - base) / (max_dist - base)
            
            # Kurangi skor jika terdeteksi VPN
            if await self.is_vpn_detected(user_id):
                score *= config["params"]["vpn_penalty"]
            
            return score
        
        elif signal_name == "off_hours":
            current_hour = datetime.now().hour
            work_start, work_end = config["params"]["work_hours"]
            
            if work_start <= current_hour < work_end:
                return 0.0  # Jam kerja
            else:
                score = 1.0
                if datetime.now().weekday() >= 5:  # Weekend
                    score += config["params"]["weekend_penalty"]
                return min(score, 1.0)
        
        elif signal_name == "rapid_sequential":
            count = await self.get_rapid_count(user_id, minutes=5)
            return 1.0 if count > config["threshold"] else 0.0
        
        return 0.0
```

**Keuntungan:**
- Tidak ada sinyal yang "menurunkan" skor
- Geografi menggunakan skor gradual, bukan binary
- VPN terdeteksi → skor dikurangi (mengurangi false positive)
- Setiap sinyal bisa di-tune secara independen

# services/security/non_api_protection_service.py

class NonApiProtectionService:
    """
    Perlindungan untuk jalur non-API.
    Semua koreksi dari review sudah diterapkan.
    """
    
    def __init__(self, db, access_service, audit, risk_scorer):
        self.db = db
        self.access_service = access_service
        self.audit = audit
        self.risk_scorer = risk_scorer
    
    # ===== 1. BACKGROUND JOBS =====
    
    async def process_extraction_job(self, job_id: str):
        """Job ekstraksi dengan service account terbatas"""
        
        # Service account dengan scope terbatas
        service_account = ServiceAccount(
            id="extraction-worker",
            scope=["document:read_pending_extraction"]
        )
        
        job = await self.db.get_job(job_id)
        
        # Access check dengan service account
        doc = await self.access_service.get_document(
            job.document_id,
            requester=service_account
        )
        
        # Ekstrak
        content = await self.extract(doc.file_path)
        
        # Index dengan access metadata
        await self.search_index.index(
            document_id=doc.id,
            content=content,
            access_metadata={
                "matter_id": doc.matter_id,
                "access_class": doc.access_class,
                "owner_organization_id": doc.owner_organization_id,
                "restricted_to": doc.restricted_to
            }
        )
        
        # Audit
        await self.audit.log(
            actor="extraction-worker",
            action="document_extracted",
            resource_id=doc.id
        )
    
    # ===== 2. DIRECT DATABASE ACCESS =====
    
    async def run_approved_script(self, script_name: str, params: dict, operator_id: str):
        """Script maintenance dengan approval workflow"""
        
        # 1. Verifikasi script ada di approved list
        if script_name not in APPROVED_SCRIPTS:
            raise SecurityError(f"Script {script_name} tidak di-approve")
        
        # 2. Minta approval
        approval = await self.request_approval(
            script_name=script_name,
            params=params,
            operator_id=operator_id,
            approver_role="security_officer"
        )
        
        if not approval.granted:
            raise ApprovalDenied("Approval tidak diberikan")
        
        # 3. Jalankan dengan audit trail
        async with AuditContext(
            actor=operator_id,
            action=f"script:{script_name}",
            reason=approval.reason
        ):
            result = await APPROVED_SCRIPTS[script_name](params)
        
        return result
    
    # ===== 3. BULK OPERATIONS =====
    
    async def bulk_export(self, matter_id: str, requester_id: str):
        """Bulk export dengan per-document access check"""
        
        # 1. Verifikasi requester adalah peserta
        participant = await self.db.get_active_participant(matter_id, requester_id)
        if not participant:
            raise AccessDenied()
        
        # 2. Ambil semua dokumen
        all_docs = await self.db.list_documents(matter_id)
        
        # 3. Filter per dokumen (TIDAK ada bulk bypass)
        accessible_docs = []
        for doc in all_docs:
            decision = await self.access_service.can_access(
                requester_id, doc.id, matter_id
            )
            if decision.granted:
                accessible_docs.append(doc)
        
        # 4. Risk scoring
        risk_score = await self.risk_scorer.calculate_risk_score(
            requester_id, "bulk_export"
        )
        
        if risk_score > 0.7:
            await self.audit.log(
                user_id=requester_id,
                action="bulk_export_high_risk",
                metadata={"risk_score": risk_score, "doc_count": len(accessible_docs)}
            )
            # Throttle, jangan block
            await asyncio.sleep(5)  # Perlambat
        
        # 5. Generate ZIP
        return await self.generate_zip(accessible_docs)
    
    # ===== 4. WEBHOOKS =====
    
    async def handle_webhook(self, webhook_type: str, payload: dict, signature: str):
        """Webhook dengan signature verification"""
        
        # 1. Verifikasi signature
        if not self.verify_signature(webhook_type, payload, signature):
            raise SecurityError("Invalid webhook signature")
        
        # 2. Validasi schema
        self.validate_schema(webhook_type, payload)
        
        # 3. Process via queue (async, bukan sync)
        await self.queue.enqueue(
            f"webhook:{webhook_type}",
            payload=payload,
            priority="low"
        )
        
        # 4. Audit
        await self.audit.log(
            action="webhook_received",
            metadata={"type": webhook_type}
        )
    
    # ===== 5. OBJECT STORAGE =====
    
    async def get_presigned_url(self, document_id: str, requester_id: str):
        """Presigned URL dengan access check di app layer"""
        
        # 1. Access check
        doc = await self.db.get_document(document_id)
        decision = await self.access_service.can_access(
            requester_id, document_id, doc.matter_id
        )
        if not decision.granted:
            raise AccessDenied()
        
        # 2. Generate presigned URL dengan expiry pendek
        url = await self.storage.generate_presigned_url(
            document_id,
            expiry_minutes=5
        )
        
        # 3. Audit
        await self.audit.log(
            user_id=requester_id,
            action="presigned_url_generated",
            resource_id=document_id,
            metadata={"expiry_minutes": 5}
        )
        
        return url
    
    # ===== 6. LOGS =====
    
    @staticmethod
    def sanitize_log(log_data: dict) -> dict:
        """Sanitasi log sebelum ditulis"""
        sensitive_fields = [
            "password", "token", "secret", "api_key",
            "document_content", "full_text", "private_key"
        ]
        
        sanitized = log_data.copy()
        for field in sensitive_fields:
            if field in sanitized:
                sanitized[field] = "***REDACTED***"
        
        return sanitized
    
    # ===== 7. CACHES =====
    
    async def get_cached_response(self, cache_key: str, requester_id: str):
        """Cache yang menghormati revocation"""
        
        cached = await self.cache.get(cache_key)
        if not cached:
            return None
        
        # Check revocation
        if await self.revocation.is_revoked_since(
            requester_id,
            cached.cached_at
        ):
            await self.cache.delete(cache_key)
            return None
        
        return cached.value
