# middleware/database_context.py

class DatabaseContextMiddleware:
    """
    Middleware yang mengelola konteks user database dengan transaksi yang benar.
    """
    
    async def __call__(self, request, call_next):
        # 1. Extract user dari token
        user = await extract_user_from_token(request)
        
        # 2. Acquire connection dari pool
        conn = await self.pool.acquire()
        
        try:
            # 3. BEGIN transaksi
            await conn.execute("BEGIN")
            
            # 4. SET LOCAL konteks user (hanya berlaku dalam transaksi ini)
            await conn.execute(
                "SET LOCAL app.current_user_id = $1",
                str(user.id)
            )
            await conn.execute(
                "SET LOCAL app.current_org_id = $1",
                str(user.organization_id)
            )
            
            # 5. Attach connection ke request
            request.state.db = conn
            
            # 6. Process request
            try:
                response = await call_next(request)
                
                # 7. COMMIT jika sukses
                await conn.execute("COMMIT")
                
                return response
            
            except Exception as e:
                # 8. ROLLBACK jika error
                await conn.execute("ROLLBACK")
                raise
        
        finally:
            # 9. Release connection
            # Catatan: Tidak perlu reset manual karena transaksi sudah selesai
            await self.pool.release(conn)


# Service layer yang menggunakan konteks

-- Trigger: Update ACL ketika peserta berubah
CREATE OR REPLACE FUNCTION update_document_acl_on_participant_change()
RETURNS TRIGGER AS $$
BEGIN
    -- Jika peserta dihapus atau status berubah
    IF TG_OP = 'DELETE' OR (TG_OP = 'UPDATE' AND OLD.status != NEW.status) THEN
        
        -- Hapus ACL untuk peserta ini
        DELETE FROM document_acl
        WHERE participant_id = COALESCE(NEW.id, OLD.id);
        
        -- Atau re-evaluate ACL berdasarkan status baru
        IF TG_OP = 'UPDATE' AND NEW.status = 'ACTIVE' THEN
            INSERT INTO document_acl (document_id, participant_id, can_access)
            SELECT 
                d.id,
                NEW.id,
                evaluate_access(d, NEW)
            FROM documents d
            WHERE d.matter_id = NEW.matter_id;
        END IF;
        
    END IF;
    
    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_participant_acl_update
AFTER INSERT OR UPDATE OR DELETE ON mediation_participants
FOR EACH ROW
EXECUTE FUNCTION update_document_acl_on_participant_change();


-- Trigger: Update ACL ketika klasifikasi dokumen berubah
CREATE OR REPLACE FUNCTION update_acl_on_document_change()
RETURNS TRIGGER AS $$
BEGIN
    -- Jika access_class atau restricted_to berubah
    IF OLD.access_class != NEW.access_class 
       OR OLD.restricted_to IS DISTINCT FROM NEW.restricted_to
       OR OLD.owner_organization_id != NEW.owner_organization_id
       OR OLD.matter_id != NEW.matter_id THEN
        
        -- Re-evaluate ACL untuk semua peserta di matter
        DELETE FROM document_acl
        WHERE document_id = NEW.id;
        
        INSERT INTO document_acl (document_id, participant_id, can_access)
        SELECT 
            NEW.id,
            mp.id,
            evaluate_access(NEW, mp)
        FROM mediation_participants mp
        WHERE mp.matter_id = NEW.matter_id
          AND mp.status = 'ACTIVE';
    
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_document_acl_update
AFTER UPDATE ON documents
FOR EACH ROW
EXECUTE FUNCTION update_acl_on_document_change();

-- Fungsi evaluate_access (dipanggil oleh trigger)
CREATE OR REPLACE FUNCTION evaluate_access(
    doc documents,
    participant mediation_participants
) RETURNS BOOLEAN AS $$
BEGIN
    -- MATTER_PUBLIC
    IF doc.access_class = 'MATTER_PUBLIC' THEN
        RETURN true;
    END IF;
    
    -- MATTER_INTERNAL
    IF doc.access_class = 'MATTER_INTERNAL' THEN
        RETURN doc.owner_organization_id = participant.organization_id;
    END IF;
    
    -- RESTRICTED
    IF doc.access_class = 'RESTRICTED' THEN
        RETURN participant.id = ANY(doc.restricted_to);
    END IF;
    
    -- MEDIATOR_EYES
    IF doc.access_class = 'MEDIATOR_EYES' THEN
        RETURN participant.role = 'MEDIATOR';
    END IF;
    
    -- COUNSEL_ONLY
    IF doc.access_class = 'COUNSEL_ONLY' THEN
        RETURN (
            (doc.counsel_party = 'CLAIMANT' AND participant.role = 'CLAIMANT_COUNSEL')
            OR
            (doc.counsel_party = 'RESPONDENT' AND participant.role = 'RESPONDENT_COUNSEL')
        );
    END IF;
    
    -- COUNSEL_AND_CLIENT
    IF doc.access_class = 'COUNSEL_AND_CLIENT' THEN
        RETURN (
            (doc.counsel_party = 'CLAIMANT' AND participant.role IN ('CLAIMANT', 'CLAIMANT_COUNSEL'))
            OR
            (doc.counsel_party = 'RESPONDENT' AND participant.role IN ('RESPONDENT', 'RESPONDENT_COUNSEL'))
        );
    END IF;
    
    -- Default: fail closed
    RETURN false;
END;
$$ LANGUAGE plpgsql;

class BulkExportService:
    async def export(self, matter_id: str, requester_id: str):
        """Bulk export dengan filter di database"""
        
        # 1. Verifikasi requester adalah peserta
        participant = await self.db.get_active_participant(matter_id, requester_id)
        if not participant:
            raise AccessDenied()
        
        # 2. Query database dengan ACL filter (bukan load semua)
        query = """
            SELECT d.*
            FROM documents d
            JOIN document_acl acl ON acl.document_id = d.id
            WHERE d.matter_id = $1
              AND acl.participant_id = $2
              AND acl.can_access = true
        """
        
        accessible_docs = await self.db.fetch(query, matter_id, participant.id)
        
        # 3. Defense in depth: verifikasi di application layer
        verified_docs = []
        for doc in accessible_docs:
            # Double check di application layer
            decision = await self.access_service.can_access(
                requester_id, doc['id'], matter_id
            )
            if decision.granted:
                verified_docs.append(doc)
            else:
                # Log anomali
                await self.audit.log(
                    action="bulk_export_acl_mismatch",
                    metadata={
                        "document_id": doc['id'],
                        "requester_id": requester_id,
                        "db_said": True,
                        "app_said": False
                    },
                    severity="CRITICAL"
                )
        
        # 4. Risk scoring
        risk_score = await self.risk_scorer.calculate_risk_score(
            requester_id, "bulk_export"
        )
        
        # 5. Jika risk tinggi, gunakan queue-based export (lihat perbaikan #6)
        if risk_score > 0.7 or len(verified_docs) > 50:
            return await self.create_export_job(matter_id, requester_id, verified_docs)
        
        # 6. Generate ZIP langsung
        return await self.generate_zip(verified_docs)

            INSERT INTO export_jobs (
                matter_id,
                requester_id,
                document_count,
                status,
                created_at
            ) VALUES ($1, $2, $3, 'PENDING', NOW())
            RETURNING *
        """, matter_id, requester_id, len(documents))

            "bulk_export",
            payload={
                "job_id": job['id'],
                "matter_id": matter_id,
                "requester_id": requester_id,
                "document_ids": [d['id'] for d in documents]
            },
            priority="low"
        )
        
        # 3. Jika risk sangat tinggi, minta approval
        risk_score = await self.risk_scorer.calculate_risk_score(
            requester_id, "bulk_export"
        )
        
        if risk_score > 0.8:
            await self.request_approval(
                job_id=job['id'],
                approver_role="security_officer",
                reason=f"Risk score: {risk_score}"
            )
        
        # 4. Return job info (bukan blocking)
        return {
            "job_id": job['id'],
            "status": "PENDING",
            "estimated_completion": "5-10 menit",
            "tracking_url": f"/exports/{job['id']}"

import re
from typing import Any

class LogSanitizer:
    """Sanitasi log dengan recursive redaction"""
    
    # Pattern untuk field sensitif (case-insensitive)
    SENSITIVE_PATTERNS = [
        r'password',
        r'token',
        r'secret',
        r'api[_-]?key',
        r'authorization',
        r'cookie',
        r'session',
        r'private[_-]?key',
        r'access[_-]?token',
        r'refresh[_-]?token',
        r'client[_-]?secret',
    ]
    
    # Allowlist: field yang BOLEH masuk log
    ALLOWLIST = {
        'user_id', 'document_id', 'matter_id', 'action', 'timestamp',
        'status', 'outcome', 'reason', 'ip_address', 'user_agent'
    }
    
    @classmethod
    def sanitize(cls, data: Any, path: str = "") -> Any:
        """Sanitasi data secara recursive"""
        
        if isinstance(data, dict):
            return {
                key: cls.sanitize(value, f"{path}.{key}" if path else key)
                for key, value in data.items()
            }
        
        elif isinstance(data, list):
            return [cls.sanitize(item, f"{path}[{i}]") for i, item in enumerate(data)]
        
        elif isinstance(data, str):
            # Check apakah ini field sensitif
            if cls._is_sensitive_field(path):
                return "***REDACTED***"
            
            # Check apakah string mengandung token/secret
            if cls._contains_sensitive_data(data):
                return "***REDACTED***"
            
            return data
        
        else:
            return data
    
    @classmethod
    def _is_sensitive_field(cls, path: str) -> bool:
        """Check apakah field name sensitif"""
        path_lower = path.lower()
        
        # Check against patterns
        for pattern in cls.SENSITIVE_PATTERNS:
            if re.search(pattern, path_lower):
                return True
        
        # Check against allowlist
        field_name = path.split('.')[-1].lower()
        if field_name in cls.ALLOWLIST:
            return False
        
        return False
    
    @classmethod
    def _contains_sensitive_data(cls, text: str) -> bool:
        """Check apakah string mengandung data sensitif"""
        
        # Pattern untuk JWT token
        if re.search(r'eyJ[A-Za-z0-9-_]+\.eyJ[A-Za-z0-9-_]+\.[A-Za-z0-9-_]+', text):
            return True
        
        # Pattern untuk API key (panjang, alphanumeric)
        if re.search(r'\b[A-Za-z0-9]{32,}\b', text):
            return True
        
        # Pattern untuk password di URL
        if re.search(r'password=[^&\s]+', text, re.IGNORECASE):
            return True
        
        return False


# Wrapper untuk logger
class SafeLogger:
    def __init__(self, logger):
        self.logger = logger
    
    def info(self, message: str, data: Any = None):
        sanitized_data = LogSanitizer.sanitize(data) if data else None
        self.logger.info(message, extra={"data": sanitized_data})
    
    def error(self, message: str, data: Any = None, exc_info: bool = False):
        sanitized_data = LogSanitizer.sanitize(data) if data else None
        self.logger.error(message, extra={"data": sanitized_data}, exc_info=exc_info)

        self,
        webhook_type: str,
        raw_body: bytes,  # Raw body, bukan parsed JSON
        headers: dict
    ):
        """Handle webhook dengan perlindungan lengkap"""
        
        # 1. Verifikasi signature dari raw body
        signature = headers.get('X-Webhook-Signature')
        if not signature:
            raise SecurityError("Missing signature")
        
        expected_signature = self._compute_signature(raw_body)
        if not hmac.compare_digest(signature, expected_signature):
            raise SecurityError("Invalid signature")
        
        # 2. Verifikasi timestamp (replay protection)
        timestamp = headers.get('X-Webhook-Timestamp')
        if not timestamp:
            raise SecurityError("Missing timestamp")
        
        try:
            webhook_time = datetime.fromisoformat(timestamp)
            now = datetime.utcnow()
            
            if abs((now - webhook_time).total_seconds()) > self.tolerance:
                raise SecurityError("Timestamp too old or too new")
        
        except ValueError:
            raise SecurityError("Invalid timestamp format")
        
        # 3. Verifikasi nonce/event ID (idempotency)
        event_id = headers.get('X-Webhook-Event-ID')
        if not event_id:
            raise SecurityError("Missing event ID")
        
        # Check apakah event sudah diproses
        if await self._is_duplicate_event(event_id):
            # Return 200 OK (idempotent)
            return {"status": "already_processed"}
        
        # 4. Verifikasi ukuran payload
        if len(raw_body) > 1_000_000:  # 1 MB
            raise SecurityError("Payload too large")
        
        # 5. Parse JSON
        try:
            payload = json.loads(raw_body)
        except json.JSONDecodeError:
            raise SecurityError("Invalid JSON")
        
        # 6. Validasi schema
        self._validate_schema(webhook_type, payload)
        
        # 7. Simpan event ID untuk idempotency
        await self._store_event_id(event_id, ttl_hours=24)
        
        # 8. Process via queue (async)
        await self.queue.enqueue(
            f"webhook:{webhook_type}",
            payload=payload,
            priority="low"
        )
        
        # 9. Audit
        await self.audit.log(
            action="webhook_processed",
            metadata={
                "type": webhook_type,
                "event_id": event_id,
                "timestamp": timestamp
            }
        )
        
        return {"status": "accepted"}
    
    def _compute_signature(self, payload: bytes) -> str:
        """Compute HMAC signature"""
        return hmac.new(
            self.secret.encode(),
            payload,
            hashlib.sha256
        ).hexdigest()
    
    async def _is_duplicate_event(self, event_id: str) -> bool:
        """Check apakah event sudah diproses"""
        result = await self.redis.get(f"webhook_event:{event_id}")
        return result is not None
    
    async def _store_event_id(self, event_id: str, ttl_hours: int):
        """Simpan event ID untuk idempotency"""
        await self.redis.setex(
            f"webhook_event:{event_id}",
            ttl_hours * 3600,
            "1"
        )
