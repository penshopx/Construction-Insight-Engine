```sql
-- Peserta mediasi dengan peran eksplisit
CREATE TABLE mediation_participants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    matter_id UUID REFERENCES matters(id) NOT NULL,
    user_id UUID REFERENCES users(id) NOT NULL,
    
    -- Peran dalam mediasi
    role VARCHAR(30) NOT NULL,
    -- 'CLAIMANT', 'RESPONDENT', 'MEDIATOR', 'OBSERVER',
    -- 'CLAIMANT_COUNSEL', 'RESPONDENT_COUNSEL', 'EXPERT_WITNESS'
    
    -- Afiliasi organisasi
    organization_id UUID REFERENCES organizations(id),
    
    -- Status
    status VARCHAR(20) DEFAULT 'ACTIVE',
    -- 'ACTIVE', 'WITHDRAWN', 'REMOVED'
    joined_at TIMESTAMPTZ DEFAULT NOW(),
    left_at TIMESTAMPTZ,
    removal_reason TEXT,
    
    -- Metadata minimal (sesuai UU PDP - data minimal yang diperlukan)
    display_name VARCHAR(100),
    contact_email VARCHAR(255),  -- dasar pemrosesan: performa kontrak
    
    -- Verifikasi identitas (opsional, sesuai kebutuhan matter)
    identity_verified BOOLEAN DEFAULT false,
    identity_verified_at TIMESTAMPTZ,
    identity_verified_by UUID REFERENCES users(id),
    
    UNIQUE(matter_id, user_id)
);

-- Dokumen dengan klasifikasi akses
CREATE TABLE documents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    matter_id UUID REFERENCES matters(id) NOT NULL,
    
    -- Klasifikasi akses (bukan kesimpulan hukum)
    access_class VARCHAR(20) NOT NULL DEFAULT 'MATTER_INTERNAL',
    -- 'MATTER_PUBLIC'    : semua peserta matter
    -- 'MATTER_INTERNAL'  : peserta dari organisasi yang sama
    -- 'RESTRICTED'       : peserta yang dieksplisitkan
    -- 'MEDIATOR_EYES'    : hanya mediator
    -- 'COUNSEL_ONLY'     : hanya counsel dari pihak peng-upload
    -- (catatan: PRIVILEGED di-rename jadi COUNSEL_ONLY untuk
    --  menghindari kesan kesimpulan hukum otomatis)
    
    -- Untuk RESTRICTED: daftar participant_id yang boleh akses
    restricted_to UUID[] DEFAULT '{}',
    
    -- Untuk COUNSEL_ONLY: pihak yang meng-upload
    counsel_party VARCHAR(20),  -- 'CLAIMANT' atau 'RESPONDENT'
    
    -- Organisasi pemilik (untuk filter MATTER_INTERNAL)
    owner_organization_id UUID REFERENCES organizations(id),
    
    -- Metadata
    uploaded_by UUID REFERENCES users(id),
    uploaded_at TIMESTAMPTZ DEFAULT NOW(),
    title TEXT,
    file_checksum VARCHAR(64) NOT NULL
);

-- View: akses user ke dokumen
-- Catatan: ini logika akses, bukan klaim hukum
CREATE VIEW user_document_access AS
SELECT 
    d.id AS document_id,
    d.matter_id,
    mp.user_id,
    mp.role,
    mp.organization_id AS participant_org_id,
    CASE 
        WHEN d.access_class = 'MATTER_PUBLIC' THEN true
        
        WHEN d.access_class = 'MATTER_INTERNAL' 
             AND d.owner_organization_id = mp.organization_id 
             THEN true
        
        WHEN d.access_class = 'RESTRICTED' 
             AND mp.id = ANY(d.restricted_to) 
             THEN true
        
        WHEN d.access_class = 'MEDIATOR_EYES' 
             AND mp.role = 'MEDIATOR' 
             THEN true
        
        WHEN d.access_class = 'COUNSEL_ONLY' 
             AND mp.role IN ('CLAIMANT_COUNSEL', 'RESPONDENT_COUNSEL')
             AND (
                 (d.counsel_party = 'CLAIMANT' AND mp.role = 'CLAIMANT_COUNSEL')
                 OR
                 (d.counsel_party = 'RESPONDENT' AND mp.role = 'RESPONDENT_COUNSEL')
             )
             THEN true
        
        ELSE false
    END AS can_access
FROM documents d
JOIN mediation_participants mp ON mp.matter_id = d.matter_id
WHERE mp.status = 'ACTIVE';

-- Row-level security (defense in depth)
-- Catatan: ini lapisan tambahan, bukan pengganti access check di app layer
ALTER TABLE documents ENABLE ROW LEVEL SECURITY;

CREATE POLICY documents_select_policy ON documents
    FOR SELECT
    USING (
        EXISTS (
            SELECT 1 FROM user_document_access uda
            WHERE uda.document_id = documents.id
              AND uda.user_id = current_setting('app.current_user_id')::uuid
              AND uda.can_access = true
        )
    );
```

### 2.3 Service Layer dengan Audit Trail

```python
# services/access/participant_access_service.py

class ParticipantAccessService:
    """
    Access check dengan audit trail.
    Catatan: kelas ini TIDAK membuat klaim hukum tentang privilege.
    Ia hanya menerapkan klasifikasi akses yang ditentukan uploader.
    """
    
    async def can_access(
        self, user_id: str, document_id: str, matter_id: str
    ) -> AccessDecision:
        
        # 1. Verifikasi user adalah peserta aktif
        participant = await self.repo.get_active_participant(matter_id, user_id)
        if not participant:
            await self.audit.log(
                user_id=user_id,
                resource_id=document_id,
                outcome="DENIED",
                reason="NOT_ACTIVE_PARTICIPANT"
            )
            return AccessDecision(granted=False)
        
        # 2. Ambil dokumen
        doc = await self.repo.get_document(document_id)
        if not doc or doc.matter_id != matter_id:
            # 404 - jangan konfirmasi keberadaan dokumen
            await self.audit.log(
                user_id=user_id,
                resource_id=document_id,
                outcome="DENIED",
                reason="DOCUMENT_NOT_FOUND_OR_NOT_IN_MATTER"
            )
            return AccessDecision(granted=False)
        
        # 3. Evaluasi klasifikasi akses
        granted = self._evaluate_access_class(doc, participant)
        
        await self.audit.log(
            user_id=user_id,
            resource_id=document_id,
            outcome="GRANTED" if granted else "DENIED",
            reason=self._last_eval_reason,
            access_class=doc.access_class,
            participant_role=participant.role,
            participant_org_id=str(participant.organization_id)
        )
        
        return AccessDecision(granted=granted)
    
    def _evaluate_access_class(self, doc, participant) -> bool:
        """Evaluasi berdasarkan klasifikasi akses."""
        
        if doc.access_class == "MATTER_PUBLIC":
            self._last_eval_reason = "MATTER_PUBLIC"
            return True
        
        if doc.access_class == "MATTER_INTERNAL":
            if doc.owner_organization_id == participant.organization_id:
                self._last_eval_reason = "SAME_ORGANIZATION"
                return True
            self._last_eval_reason = "CROSS_ORGANIZATION"
            return False
        
        if doc.access_class == "RESTRICTED":
            if participant.id in (doc.restricted_to or []):
                self._last_eval_reason = "EXPLICITLY_IN_RESTRICTED_LIST"
                return True
            self._last_eval_reason = "NOT_IN_RESTRICTED_LIST"
            return False
        
        if doc.access_class == "MEDIATOR_EYES":
            if participant.role == "MEDIATOR":
                self._last_eval_reason = "MEDIATOR_ROLE"
                return True
            self._last_eval_reason = "NOT_MEDIATOR_ROLE"
            return False
        
        if doc.access_class == "COUNSEL_ONLY":
            is_matching_counsel = (
                (doc.counsel_party == "CLAIMANT" and participant.role == "CLAIMANT_COUNSEL")
                or
                (doc.counsel_party == "RESPONDENT" and participant.role == "RESPONDENT_COUNSEL")
            )
            if is_matching_counsel:
                self._last_eval_reason = "MATCHING_COUNSEL_ROLE"
                return True
            self._last_eval_reason = "NOT_MATCHING_COUNSEL"
            return False
        
        # Unknown class → fail closed
        self._last_eval_reason = "UNKNOWN_ACCESS_CLASS"
        return False
```

```python
# services/security/non_api_protection.py

class NonApiProtectionService:
    """
    Perlindungan untuk jalur non-API.
    Prinsip: access check TIDAK BOLEH di-bypass oleh jalur apapun.
    """
    
    # ===== 1. BACKGROUND JOBS =====
    
    async def process_document_extraction(self, job_id: str):
        """
        Job ekstraksi TIDAK BOLEH membaca dokumen tanpa access check.
        Gunakan service account dengan scope terbatas.
        """
        job = await self.repo.get_job(job_id)
        
        # Service account khusus untuk ekstraksi
        # Scope: hanya dokumen yang statusnya "PENDING_EXTRACTION"
        # TIDAK bisa akses dokumen lain
        service_account = ServiceAccount(
            id="extraction-worker",
            scope=["document:read_pending_extraction"]
        )
        
        # Access check tetap berlaku
        doc = await self.access_service.get_document(
            job.document_id, 
            requester=service_account
        )
        
        # Ekstrak konten
        content = await self.extractor.extract(doc.file_path)
        
        # Index ke search — dengan access metadata
        await self.search_index.index(
            document_id=doc.id,
            content=content,
            access_metadata={
                "matter_id": doc.matter_id,
                "access_class": doc.access_class,
                "owner_organization_id": doc.owner_organization_id,
                "restricted_to": doc.restricted_to
            }
        )
    
    # ===== 2. DIRECT DATABASE ACCESS =====
    
    # Kebijakan: TIDAK ADA direct DB access di production
    # Semua operasi via API atau approved scripts dengan audit
    
    @classmethod
    async def run_approved_script(cls, script_name: str, params: dict):
        """
        Script maintenance harus melalui approval workflow.
        Semua aksi dicatat di audit log terpisah.
        """
        # 1. Verifikasi script ada di approved list
        if script_name not in APPROVED_SCRIPTS:
            raise SecurityError("Script not approved")
        
        # 2. Minta approval dari security officer
        approval = await cls.request_approval(
            script_name=script_name,
            params=params,
            approver="security_officer"
        )
        
        # 3. Jalankan dengan audit trail lengkap
        async with AuditContext(
            actor="script_runner",
            action=f"script:{script_name}",
            reason=approval.reason
        ):
            result = await APPROVED_SCRIPTS[script_name](params)
        
        return result
    
    # ===== 3. BULK OPERATIONS =====
    
    async def bulk_export(self, matter_id: str, requester_id: str):
        """
        Bulk export TETAP harus honor per-document access check.
        Tidak ada "bulk bypass".
        """
        # 1. Verifikasi requester adalah peserta matter
        participant = await self.repo.get_active_participant(matter_id, requester_id)
        if not participant:
            raise AccessDenied()
        
        # 2. Ambil SEMUA dokumen di matter
        all_docs = await self.repo.list_documents(matter_id)
        
        # 3. Filter berdasarkan access check PER DOKUMEN
        accessible_docs = []
        for doc in all_docs:
            decision = await self.access_service.can_access(
                requester_id, doc.id, matter_id
            )
            if decision.granted:
                accessible_docs.append(doc)
            # Catatan: dokumen yang tidak bisa diakses TIDAK masuk export,
            # dan TIDAK ada indikasi bahwa dokumen itu ada
        
        # 4. Rate limit bulk export
        await self.rate_limiter.check_bulk_export(requester_id)
        
        # 5. Generate ZIP
        return await self.generate_zip(accessible_docs)
    
    # ===== 4. WEBHOOKS =====
    
    async def handle_webhook(self, webhook_type: str, payload: dict):
        """
        Webhook harus divalidasi signature-nya.
        Webhook TIDAK BOLEH langsung menulis data sensitif.
        """
        # 1. Verifikasi signature
        if not self.verify_webhook_signature(webhook_type, payload):
            raise SecurityError("Invalid webhook signature")
        
        # 2. Validasi payload schema
        self.validate_payload_schema(webhook_type, payload)
        
        # 3. Proses via queue (bukan langsung)
        # Ini mencegah webhook jadi vektor serangan synchronous
        await self.queue.enqueue(
            f"webhook:{webhook_type}",
            payload=payload,
            priority="low"
        )
    
    # ===== 5. OBJECT STORAGE =====
    
    async def get_presigned_url(self, document_id: str, requester_id: str):
        """
        Presigned URL untuk S3/MinIO TIDAK BOLEH di-generate
        tanpa access check di application layer.
        """
        # 1. Access check di app layer
        doc = await self.repo.get_document(document_id)
        decision = await self.access_service.can_access(
            requester_id, document_id, doc.matter_id
        )
        if not decision.granted:
            raise AccessDenied()
        
        # 2. Generate presigned URL dengan expiry pendek
        # Catatan: URL ini bisa di-share, jadi expiry harus pendek
        url = await self.storage.generate_presigned_url(
            document_id,
            expiry_minutes=5  # Pendek untuk memitigasi risiko share
        )
        
        # 3. Audit log
        await self.audit.log(
            user_id=requester_id,
            action="presigned_url_generated",
            resource_id=document_id,
            metadata={"expiry_minutes": 5}
        )
        
        return url
    
    # ===== 6. LOGS =====
    
    @classmethod
    def sanitize_log(cls, log_data: dict) -> dict:
        """
        Sanitasi log sebelum ditulis.
        Hapus/mask data sensitif.
        """
        sensitive_fields = [
            "password", "token", "secret", "api_key",
            "document_content", "full_text"
        ]
        
        sanitized = log_data.copy()
        for field in sensitive_fields:
            if field in sanitized:
                sanitized[field] = "***REDACTED***"
        
        return sanitized
    
    # ===== 7. CACHES =====
    
    async def get_cached_response(self, cache_key: str, requester_id: str):
        """
        Cached response harus honor revocation.
        Cache TIDAK BOLEH menyimpan data yang sudah di-revoke.
        """
        cached = await self.cache.get(cache_key)
        
        if not cached:
            return None
        
        # Verifikasi cache masih valid
        # (tidak ada revocation sejak cache dibuat)
        if await self.revocation.is_revoked_since(
            requester_id, 
            cached.cached_at
        ):
            await self.cache.delete(cache_key)
            return None
        
        return cached.value
```

```python
# services/rate_limiting/rate_limiter.py

class RiskBasedRateLimiter:
    """
    Rate limiting berbasis sinyal risiko.
    Bukan blocking mutlak, tapi eskalasi bertahap.
    """
    
    # Layer 1: Hard limits (untuk mencegah abuse jelas)
    HARD_LIMITS = {
        "document_download_per_hour": 100,
        "bulk_export_per_day": 5,
        "api_calls_per_minute": 300,
        "login_attempts_per_hour": 10,
    }
    
    # Layer 2: Risk signals (untuk deteksi anomali)
    RISK_SIGNALS = {
        "unusual_download_volume": {
            "threshold": "3x rata-rata harian user",
            "action": "flag_for_review"
        },
        "new_device_access": {
            "threshold": "perangkat belum pernah dipakai",
            "action": "require_additional_verification"
        },
        "geographic_anomaly": {
            "threshold": "lokasi >500km dari pola biasa",
            "action": "flag_for_review"
        },
        "off_hours_bulk_access": {
            "threshold": "bulk download di luar jam kerja",
            "action": "flag_for_review"
        },
        "rapid_sequential_access": {
            "threshold": ">50 dokumen dalam 5 menit",
            "action": "temporary_throttle"
        }
    }
    
    async def check_download(self, user_id: str, document_id: str):
        """Check download dengan multi-layer"""
        
        # Layer 1: Hard limit
        count = await self.get_count(f"download:{user_id}:hour")
        if count >= self.HARD_LIMITS["document_download_per_hour"]:
            await self.audit.log(
                user_id=user_id,
                action="download_blocked_hard_limit",
                metadata={"count": count}
            )
            raise RateLimitExceeded("Download limit tercapai. Coba lagi nanti.")
        
        # Layer 2: Risk signals
        risk_score = await self.calculate_risk_score(user_id, "download")
        
        if risk_score > 0.8:  # High risk
            await self.audit.log(
                user_id=user_id,
                action="download_flagged_high_risk",
                metadata={"risk_score": risk_score}
            )
            # Jangan block, tapi:
            # 1. Perlambat response (throttle)
            # 2. Notifikasi ke security team
            # 3. Minta verifikasi tambahan untuk download berikutnya
            await self.notify_security_team(user_id, risk_score)
            await self.require_additional_verification(user_id)
            return DownloadDecision(
                allowed=True,
                throttled=True,
                requires_verification=True
            )
        
        elif risk_score > 0.5:  # Medium risk
            await self.audit.log(
                user_id=user_id,
                action="download_flagged_medium_risk",
                metadata={"risk_score": risk_score}
            )
            return DownloadDecision(
                allowed=True,
                throttled=False,
                requires_verification=False
            )
        
        # Low risk
        return DownloadDecision(
            allowed=True,
            throttled=False,
            requires_verification=False
        )
    
    async def calculate_risk_score(self, user_id: str, action: str) -> float:
        """
        Hitung risk score berdasarkan sinyal.
        Return: 0.0 (low risk) - 1.0 (high risk)
        """
        signals = []
        
        # Sinyal 1: Volume tidak biasa
        avg_daily = await self.get_average_daily(user_id, action)
        current = await self.get_current_count(user_id, action, "day")
        if current > avg_daily * 3:
            signals.append(0.4)
        
        # Sinyal 2: Perangkat baru
        if await self.is_new_device(user_id):
            signals.append(0.3)
        
        # Sinyal 3: Lokasi tidak biasa
        if await self.is_geographic_anomaly(user_id):
            signals.append(0.3)
        
        # Sinyal 4: Di luar jam kerja
        if await self.is_off_hours(user_id):
            signals.append(0.2)
        
        # Sinyal 5: Akses cepat berturut-turut
        if await self.is_rapid_sequential(user_id, action):
            signals.append(0.3)
        
        # Combine signals (bobot bisa di-tune)
        if not signals:
            return 0.0
        
        # Weighted average dengan cap di 1.0
        combined = sum(signals) / len(signals) * 1.5
        return min(combined, 1.0)
    
    async def recovery_process(self, user_id: str):
        """
        Proses pemulihan untuk user yang di-flag.
        Bukan blocking permanen, tapi verifikasi.
        """
        # 1. Notifikasi ke user
        await self.notify_user(
            user_id,
            message="Akun Anda terdeteksi aktivitas tidak biasa. "
                   "Silakan verifikasi identitas untuk melanjutkan."
        )
        
        # 2. Minta verifikasi tambahan
        # Opsi: email confirmation, SMS OTP, atau contact support
        
        # 3. Setelah verifikasi, reset risk score
        await self.reset_risk_score(user_id)
        
        # 4. Audit log
        await self.audit.log(
            user_id=user_id,
            action="risk_recovery_completed"
        )
```

### 6.2 Blueprint FASE 2 (Enkripsi Terbatas)

```python
# services/encryption/counsel_only_encryption.py

class CounselOnlyEncryptionService:
    """
    Enkripsi terbatas untuk dokumen COUNSEL_ONLY.
    Platform tidak bisa decrypt tanpa persetujuan.
    """
    
    async def encrypt_document(self, document: Document, counsel_key: PublicKey, client_key: PublicKey):
        """
        Encrypt dokumen dengan public key counsel + client.
        Dokumen hanya bisa di-decrypt dengan private key salah satu.
        """
        # Generate symmetric key untuk dokumen
        doc_key = generate_symmetric_key()
        
        # Encrypt dokumen dengan symmetric key
        encrypted_content = encrypt_aes(document.content, doc_key)
        
        # Encrypt symmetric key dengan public key counsel
        encrypted_key_for_counsel = encrypt_rsa(doc_key, counsel_key)
        
        # Encrypt symmetric key dengan public key client
        encrypted_key_for_client = encrypt_rsa(doc_key, client_key)
        
        # Simpan
        await self.storage.save(
            document_id=document.id,
            encrypted_content=encrypted_content,
            encrypted_keys={
                "counsel": encrypted_key_for_counsel,
                "client": encrypted_key_for_client
            }
        )
        
        # Metadata (tidak di-encrypt, untuk search)
        await self.metadata.save(
            document_id=document.id,
            title=document.title,
            access_class=document.access_class,
            matter_id=document.matter_id,
            # Catatan: konten TIDAK di-index untuk search
        )
    
    async def decrypt_document(self, document_id: str, requester_private_key: PrivateKey):
        """
        Decrypt dokumen dengan private key requester.
        Requester harus counsel atau client.
        """
        # Ambil encrypted data
        encrypted = await self.storage.load(document_id)
        
        # Tentukan encrypted key mana yang dipakai
        requester_role = await self.get_requester_role(document_id)
        
        if requester_role == "COUNSEL":
            encrypted_key = encrypted.encrypted_keys["counsel"]
        elif requester_role == "CLIENT":
            encrypted_key = encrypted.encrypted_keys["client"]
        else:
            raise AccessDenied("Bukan counsel atau client")
        
        # Decrypt symmetric key
        doc_key = decrypt_rsa(encrypted_key, requester_private_key)
        
        # Decrypt konten
        content = decrypt_aes(encrypted.encrypted_content, doc_key)
        
        # Audit log
        await self.audit.log(
            action="counsel_only_document_decrypted",
            document_id=document_id,
            metadata={"requester_role": requester_role}
        )
        
        return content
    
    async def search(self, query: str, user_id: str):
        """
        Search untuk COUNSEL_ONLY documents.
        Hanya metadata, bukan konten.
        """
        # Search hanya di metadata (title, description)
        # Konten TIDAK bisa di-search (karena encrypted)
        results = await self.metadata.search(query)
        
        # Filter berdasarkan access
        filtered = []
        for doc in results:
            if await self.access_service.can_access(user_id, doc.id, doc.matter_id):
                filtered.append(doc)
        
        return filtered
    
    async def court_order_decrypt(self, document_id: str, court_order: CourtOrder):
        """
        Decrypt untuk court order.
        Perlu escrow key + audit trail ketat.
        """
        # 1. Verifikasi court order
        if not self.verify_court_order(court_order):
            raise InvalidCourtOrder()
        
        # 2. Gunakan escrow key (hanya untuk court order)
        escrow_key = await self.get_escrow_key()
        
        # 3. Decrypt
        content = await self._decrypt_with_key(document_id, escrow_key)
        
        # 4. Audit log ketat
        await self.audit.log(
            action="court_order_decrypt",
            document_id=document_id,
            metadata={
                "court_order_id": court_order.id,
                "court": court_order.court,
                "case_number": court_order.case_number,
                "verified_by": court_order.verified_by
            },
            severity="CRITICAL"
        )
        
        # 5. Notifikasi ke counsel + client
        await self.notify_parties(document_id, "court_order_executed")
        
        return content
```
